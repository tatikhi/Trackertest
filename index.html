        // ==========================================

        // Ниже в массиве заданий (строки примерно 345-365) у каждого дня есть поля "image" и "audio".

        // ВАРИАНТ А (Самый удобный): Вы открываете сайт в браузере и прямо на экране нажимаете кнопку "Выбрать файл" под каждым днем (сайт сам превратит их в нужный формат).

        // ВАРИАНТ Б (Жестко прописать ссылки в код): Вы можете прямо в кавычках полей image: "" и audio: "" вставить прямую ссылку на картинку/аудио в интернете или путь к файлу на GitHub (например, "images/day1.jpg").

        // ==========================================

 

        const defaultLessons = [

            {

                id: 1,

                topic: "День 1: «Охота за цветами» (Цвета)",

                desc: "🏃‍♂️ Задание: Бегом найди в комнате что-то КРАСНОЕ, схвати предмет, покажи в камеру и громко крикни: «Red!»",

                status: "Запланировано",

                image: "", // <--- СЮДА (или через кнопку на сайте) вставляется картинка для Дня 1

                audio: ""  // <--- СЮДА (или через кнопку на сайте) вставляется аудио для Дня 1

            },

            {

                id: 2,

                topic: "День 2: «Супер-робот» (Команды тела)",

                desc: "🤖 Задание: Включи режим робота! Послушай аудио, потопай ногами под «Stomp!» и покажи язык под «Show me your tongue!»",

                status: "Запланировано",

                image: "", // <--- СЮДА (или через кнопку на сайте) вставляется картинка для Дня 2

                audio: ""  // <--- СЮДА (или через кнопку на сайте) вставляется аудио для Дня 2

            }

        ];

 

        let lessons = [];

 

        // Кодирование/декодирование данных для URL

        function encodeData(data) {

            return btoa(encodeURIComponent(JSON.stringify(data)));

        }

 

        function decodeData(str) {

            try {

                return JSON.parse(decodeURIComponent(atob(str)));

            } catch (e) {

                return null;

            }

        }

 

        function initData() {

            const hash = window.location.hash;

            if (hash && hash.startsWith('#data=')) {

                const encodedStr = hash.replace('#data=', '');

                const decoded = decodeData(encodedStr);

                if (decoded) {

                    lessons = decoded;

                    localStorage.setItem('english_2days_data', JSON.stringify(lessons));

                    return;

                }

            }

            lessons = JSON.parse(localStorage.getItem('english_2days_data')) || defaultLessons;

        }

 

        function saveState() {

            localStorage.setItem('english_2days_data', JSON.stringify(lessons));

            updateProgress();

        }

 

        function updateProgress() {

            const completedCount = lessons.filter(l => l.status === "Пройдено").length;

            const percentage = Math.round((completedCount / lessons.length) * 100);

           

            document.getElementById('progress-text').innerText = `${completedCount} / ${lessons.length} (${percentage}%)`;

            document.getElementById('progress-bar').style.width = `${percentage}%`;

        }

 

        function toggleStatus(id) {

            const lesson = lessons.find(l => l.id === id);

            if (lesson) {

                lesson.status = lesson.status === "Запланировано" ? "Пройдено" : "Запланировано";

                saveState();

                renderLessons();

            }

        }

 

        // Загрузка локальных файлов (картинки и аудио) через кнопки на сайте

        function handleFileUpload(id, type, event) {

            const file = event.target.files[0];

            if (!file) return;

 

            const reader = new FileReader();

            reader.onload = function(e) {

                const lesson = lessons.find(l => l.id === id);

                if (lesson) {

                    lesson[type] = e.target.result;

                    saveState();

                    renderLessons();

                }

            };

            reader.readAsDataURL(file);

        }

 

        function copyShareableLink() {

            const encoded = encodeData(lessons);

            const shareUrl = `${window.location.origin}${window.location.pathname}#data=${encoded}`;

           

            navigator.clipboard.writeText(shareUrl).then(() => {

                showToast("Ссылка с актуальным прогрессом скопирована!");

            }).catch(() => {

                showToast("Не удалось скопировать ссылку");

            });

        }

 

        function showToast(message) {

            const toast = document.getElementById('toast');

            toast.innerText = message;

            toast.classList.add('show');

            setTimeout(() => {

                toast.classList.remove('show');

            }, 3000);

        }

 

        function renderLessons() {

            const container = document.getElementById('lessons-container');

            container.innerHTML = '';

 

            lessons.forEach(lesson => {

                const card = document.createElement('div');

                card.className = 'lesson-card';

 

                const isCompleted = lesson.status === "Пройдено";

                const badgeClass = isCompleted ? "status-completed" : "status-planned";

 

                card.innerHTML = `

                    <div class="lesson-header">

                        <span class="lesson-number">Миссия ${lesson.id}</span>

                        <span class="status-badge ${badgeClass}" onclick="toggleStatus(${lesson.id})">

                            ${lesson.status === "Запланировано" ? "⏳ Ожидает выполнения" : "✅ Пройдено!"}

                        </span>

                    </div>

 

                    <div class="lesson-topic">${lesson.topic}</div>

                    <div class="lesson-desc">${lesson.desc}</div>

 

                    <div class="media-box">

                        <label>🖼 Картинка-иллюстрация к заданию:</label>

                        <input type="file" accept="image/*" onchange="handleFileUpload(${lesson.id}, 'image', event)">

                        <img class="preview-image" src="${lesson.image}" style="display: ${lesson.image ? 'block' : 'none'};">

                    </div>

 

                    <div class="media-box">

                        <label>🎙 Ваше голосовое сообщение (аудио):</label>

                        <input type="file" accept="audio/*" onchange="handleFileUpload(${lesson.id}, 'audio', event)">

                        ${lesson.audio ? `<audio controls src="${lesson.audio}"></audio>` : ''}

                    </div>

 

                    <div class="card-footer">

                        <a href="https://t.me/tati_khi?text=${encodeURIComponent('Здравствуйте! Отправляю видео-отчет за Миссию ' + lesson.id)}" target="_blank" class="tg-btn">

                            ✈️ Отправить видео-отчет в Telegram

                        </a>

                        <span style="font-size: 0.8rem; color: #64748b;">(Или отметьте галочкой выше)</span>

                    </div>

                `;

 

                container.appendChild(card);

            });

        }

 

        // Запуск при загрузке

        initData();

        renderLessons();

        updateProgress();

    </script>
